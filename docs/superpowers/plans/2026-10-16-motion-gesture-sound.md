# Motion, Gestures, Haptics and Sound Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give the Claritel Agent app consistent motion, gestures, haptics and UI sounds, all routed through one semantic feedback module.

**Architecture:** Screens call `feedback('event')`. A pure `decide()` function maps each event to a haptic and a sound, after checking the user's settings, whether a call is live, and whether the app is running on web. Motion uses small in-house primitives built on Reanimated 4 and Gesture Handler 2: `PressableScale`, `AnimatedSheet`, `SwipeRow`, `SlideToAnswer`, `FadeInItem` and `RollingDigits`. They share timing and spring tokens in `src/theme/motion.ts` and honour the OS Reduce Motion setting. Sounds are synthesized WAV placeholders, generated by a Node script and played through `expo-audio` in a mode that respects the silent switch.

**Tech Stack:** Expo SDK 57, Expo Router, React Native 0.86, react-native-reanimated 4.5 with react-native-worklets 0.10, react-native-gesture-handler 2.32, expo-audio 57, expo-haptics, zustand 5, jest-expo.

**Spec:** `docs/superpowers/specs/2026-10-16-motion-gesture-sound-design.md`

```ts
import { Easing, withSpring, withTiming, type WithSpringConfig } from 'react-native-reanimated';

export const duration = { fast: 150, base: 220, slow: 320 } as const;

// Close to critically damped: a slight settle, no visible bounce.
export const spring = {
  snappy: { damping: 34, stiffness: 420, mass: 1 },
  gentle: { damping: 29, stiffness: 260, mass: 1 },
} satisfies Record<string, WithSpringConfig>;

export const easing = {
  enter: Easing.out(Easing.cubic),
  exit: Easing.in(Easing.cubic),
};

/** Spring to `to`, or a short timed move when Reduce Motion is on. */
export function animateTo(to: number, kind: keyof typeof spring, reduced: boolean) {
  'worklet';
  return reduced ? withTiming(to, { duration: duration.fast }) : withSpring(to, spring[kind]);
}
```

- [ ] **Step 4: Create `src/feedback/gate.ts`**

```ts
import { dtmfSound, events, type FeedbackEvent, type HapticKind, type SoundName } from './events';

export type GateContext = { hapticsOn: boolean; soundsOn: boolean; inCall: boolean; web: boolean };
export type Decision = { haptic: HapticKind | null; sound: SoundName | null };

export function decide(event: FeedbackEvent, ctx: GateContext, digit?: string): Decision {
  const e = events[event];
  const haptic = e.haptic && ctx.hapticsOn && !ctx.web ? e.haptic : null;
  let sound: SoundName | null = null;
  if (e.sound && ctx.soundsOn && (!ctx.inCall || e.critical)) {
    sound = e.sound === 'dtmf' ? (digit ? dtmfSound(digit) : null) : e.sound;
  }
  return { haptic, sound };
}
```

```js
// Synthesizes placeholder UI sounds. Run: npm run sounds
// Replace any file in assets/sounds with a branded one of the same name.
import { mkdirSync, writeFileSync } from 'node:fs';
import { dirname, join } from 'node:path';
import { fileURLToPath } from 'node:url';

const RATE = 44100;
const out = join(dirname(fileURLToPath(import.meta.url)), '..', 'assets', 'sounds');
mkdirSync(out, { recursive: true });

// A segment is a chord of sine waves. freqs: [] is silence.
// gain ~0.2 peak keeps tones around -18 LUFS and quiet beside call audio.
function render(segments) {
  const chunks = segments.map((seg) => {
    const n = Math.round((RATE * seg.ms) / 1000);
    const fade = Math.max(1, Math.min(Math.round(RATE * 0.008), Math.floor(n / 2)));
    const buf = new Float32Array(n);
    for (let i = 0; i < n; i++) {
      const t = i / RATE;
      let v = 0;
      for (const f of seg.freqs) v += Math.sin(2 * Math.PI * f * t);
      v = seg.freqs.length ? v / seg.freqs.length : 0;
      const env = Math.min(1, i / fade, (n - 1 - i) / fade);
      const decay = seg.decay ? Math.exp(-t * seg.decay) : 1;
      buf[i] = v * env * decay * (seg.gain ?? 0.2);
    }
    return buf;
  });
  const total = chunks.reduce((a, c) => a + c.length, 0);
  const all = new Float32Array(total);
  let o = 0;
  for (const c of chunks) {
    all.set(c, o);
    o += c.length;
  }
  return all;
}

function wav(samples) {
  const data = Buffer.alloc(samples.length * 2);
  samples.forEach((s, i) => data.writeInt16LE(Math.round(Math.max(-1, Math.min(1, s)) * 32767), i * 2));
  const h = Buffer.alloc(44);
  h.write('RIFF', 0);
  h.writeUInt32LE(36 + data.length, 4);
  h.write('WAVE', 8);
  h.write('fmt ', 12);
  h.writeUInt32LE(16, 16); // fmt chunk size
  h.writeUInt16LE(1, 20); // PCM
  h.writeUInt16LE(1, 22); // mono
  h.writeUInt32LE(RATE, 24);
  h.writeUInt32LE(RATE * 2, 28); // byte rate
  h.writeUInt16LE(2, 32); // block align
  h.writeUInt16LE(16, 34); // bits per sample
  h.write('data', 36);
  h.writeUInt32LE(data.length, 40);
  return Buffer.concat([h, data]);
}

const gap = { freqs: [], ms: 20 };

// Standard DTMF pairs (row Hz, column Hz)
const dtmf = {
  1: [697, 1209],
  2: [697, 1336],
  3: [697, 1477],
  4: [770, 1209],
  5: [770, 1336],
  6: [770, 1477],
  7: [852, 1209],
  8: [852, 1336],
  9: [852, 1477],
  star: [941, 1209],
  0: [941, 1336],
  hash: [941, 1477],
};

const sounds = {
  ...Object.fromEntries(Object.entries(dtmf).map(([k, freqs]) => [`dtmf_${k}`, [{ freqs, ms: 120, gain: 0.2 }]])),
  connect: [{ freqs: [660], ms: 90 }, gap, { freqs: [880], ms: 140, decay: 8 }],
  end: [{ freqs: [880], ms: 90 }, gap, { freqs: [520], ms: 160, decay: 6 }],
  saved: [{ freqs: [784], ms: 70, gain: 0.16 }, gap, { freqs: [1175], ms: 180, decay: 10, gain: 0.16 }],
  complete: [{ freqs: [1046, 2093], ms: 160, decay: 14, gain: 0.16 }],
  error: [{ freqs: [330, 349], ms: 220, decay: 5, gain: 0.18 }],
};

for (const [name, segs] of Object.entries(sounds)) {
```

```ts
import * as Haptics from 'expo-haptics';
import type { HapticKind } from './events';

const rank: Record<HapticKind, number> = { selection: 0, light: 1, medium: 2, warning: 3, success: 3, error: 4 };
let lastAt = 0;
let lastRank = -1;

function run(kind: HapticKind): Promise<void> {
  switch (kind) {
    case 'selection':
      return Haptics.selectionAsync();
    case 'light':
      return Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light);
    case 'medium':
      return Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Medium);
    case 'success':
      return Haptics.notificationAsync(Haptics.NotificationFeedbackType.Success);
    case 'warning':
      return Haptics.notificationAsync(Haptics.NotificationFeedbackType.Warning);
    case 'error':
      return Haptics.notificationAsync(Haptics.NotificationFeedbackType.Error);
  }
}

/**
 * Fire-and-forget. A button's light tap followed in the same tick by a stronger
 * event (e.g. success) plays both; an equal or weaker haptic within 60 ms is dropped.
 */
export function playHaptic(kind: HapticKind) {
  const now = Date.now();
  if (now - lastAt < 60 && rank[kind] <= lastRank) return;
  lastAt = now;
  lastRank = rank[kind];
  try {
    void run(kind).catch(() => undefined);
  } catch {
    // Haptics unavailable on this device
  }
}
```

```ts
    warned = true;

  }

/** Call once after startup. Safe to call again; failures never throw. */
export async function preloadSounds() {
  if (started) return;
  started = true;
  try {
    // Respect the silent switch and never take audio focus from call audio.
    // When the real VoIP stack lands, re-check this against its audio session.
    await setAudioModeAsync({ playsInSilentMode: false, interruptionMode: 'mixWithOthers' });
  } catch (e) {
    warnOnce(e);
  }
  for (const [name, src] of Object.entries(sources) as [SoundName, number][]) {
    try {
      players.set(name, createAudioPlayer(src));
    } catch (e) {
      warnOnce(e);
    }
  }
}

export function playSound(name: SoundName) {
  const p = players.get(name);
  if (!p) return;
  try {
    void p
      .seekTo(0)
      .then(() => p.play())
      .catch(warnOnce);
  } catch (e) {
    warnOnce(e);
  }
}
```

```tsx
  style?: StyleProp<ViewStyle>;

      {...rest}
      disabled={disabled}

        onPressOut?.(e);
```

```tsx
import { useEffect, useState, type ReactNode } from 'react';
import { KeyboardAvoidingView, Modal, Platform, Pressable, useWindowDimensions, View, type StyleProp, type ViewStyle } from 'react-native';
import { Gesture, GestureDetector, GestureHandlerRootView } from 'react-native-gesture-handler';
import Animated, {
  Extrapolation,
  interpolate,
  runOnJS,
  useAnimatedStyle,
  useReducedMotion,
  useSharedValue,
  withTiming,
} from 'react-native-reanimated';
import { feedback } from '@/feedback';
import { useTheme } from '@/theme';
import { animateTo, duration, easing } from '@/theme/motion';

const DISMISS = 120; // px dragged down
const FLING = 900; // px/s downward velocity

export function AnimatedSheet({
  open,
  onClose,
  handle,
  children,
  sheetStyle,
}: {
  open: boolean;
  onClose: () => void;
  /** The draggable area: grabber and title. Content below it scrolls normally. */
  handle: ReactNode;
  children: ReactNode;
  sheetStyle?: StyleProp<ViewStyle>;
}) {
  const { c } = useTheme();
  const { height } = useWindowDimensions();
  const reduced = useReducedMotion();
  const [mounted, setMounted] = useState(open);
  const y = useSharedValue(height);
  const armed = useSharedValue(false);

  useEffect(() => {
    if (open) {
      setMounted(true);
      // Starts from wherever y is, so reopening mid-close is interruptible.
      y.value = animateTo(0, 'gentle', reduced);
    } else {
      y.value = withTiming(height, { duration: reduced ? duration.fast : duration.base, easing: easing.exit }, (done) => {
        if (done) runOnJS(setMounted)(false);
      });
    }
    // Only react to open/close. Re-running on `mounted` would restart the enter animation.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [open]);

  const pan = Gesture.Pan()
    .onUpdate((e) => {
      // Rubber band when pulled up past the top
      y.value = e.translationY > 0 ? e.translationY : e.translationY / 4;
      const past = e.translationY > DISMISS;
      if (past !== armed.value) {
        armed.value = past;
        if (past) runOnJS(feedback)('sheet.snap');
      }
    })
    .onEnd((e) => {
      armed.value = false;
      if (e.translationY > DISMISS || e.velocityY > FLING) runOnJS(onClose)();
      else y.value = animateTo(0, 'gentle', reduced);
    });

  const sheetAnim = useAnimatedStyle(() => ({ transform: [{ translateY: Math.max(y.value, -24) }] }));
  const backdropAnim = useAnimatedStyle(() => ({ opacity: interpolate(y.value, [0, height], [1, 0], Extrapolation.CLAMP) }));

  if (!mounted) return null;

  return (
    <Modal visible transparent animationType="none" onRequestClose={onClose} statusBarTranslucent>
      {/* Android Modals render outside the root view and need their own gesture root */}
      <GestureHandlerRootView style={{ flex: 1 }}>
        <KeyboardAvoidingView behavior={Platform.OS === 'ios' ? 'padding' : undefined} style={{ flex: 1, justifyContent: 'flex-end' }}>
          <Animated.View style={[{ position: 'absolute', inset: 0, backgroundColor: c.overlay }, backdropAnim]}>
            <Pressable accessibilityLabel="Close" onPress={onClose} style={{ flex: 1 }} />
          </Animated.View>
          <Animated.View style={[sheetStyle, sheetAnim]}>
            <GestureDetector gesture={pan}>
              <View>{handle}</View>
            </GestureDetector>
            {children}
          </Animated.View>
        </KeyboardAvoidingView>
      </GestureHandlerRootView>
    </Modal>
  );
}
```

```tsx
import type { ReactNode } from 'react';
import { View } from 'react-native';
import { Gesture, GestureDetector } from 'react-native-gesture-handler';
import Animated, { Extrapolation, interpolate, runOnJS, useAnimatedStyle, useReducedMotion, useSharedValue } from 'react-native-reanimated';
import { Icon, type IconName } from '@/components/ui';
import { feedback } from '@/feedback';
import { animateTo } from '@/theme/motion';

export type SwipeAction = { icon: IconName; bg: string; fg: string; onTrigger: () => void };

const THRESHOLD = 84;
const MAX = THRESHOLD * 1.3;

/**
 * `left` is revealed on the left edge and fires on a swipe RIGHT; `right` fires on a swipe LEFT.
 * Give the child Pressable matching accessibilityActions; swipes are not reachable by screen readers.
 */
export function SwipeRow({ left, right, children }: { left?: SwipeAction; right?: SwipeAction; children: ReactNode }) {
  const reduced = useReducedMotion();
  const x = useSharedValue(0);
  const zone = useSharedValue(0); // -1 | 0 | 1
  const hasLeft = !!left;
  const hasRight = !!right;
  const fireLeft = () => left?.onTrigger();
  const fireRight = () => right?.onTrigger();

  const pan = Gesture.Pan()
    .enabled(hasLeft || hasRight)
    .activeOffsetX([-12, 12])
    .failOffsetY([-10, 10])
    .onUpdate((e) => {
      x.value = Math.min(hasLeft ? MAX : 0, Math.max(hasRight ? -MAX : 0, e.translationX));
      const z = x.value > THRESHOLD ? 1 : x.value < -THRESHOLD ? -1 : 0;
      if (z !== zone.value) {
        zone.value = z;
        if (z !== 0) runOnJS(feedback)('swipe.threshold');
      }
    })
    .onEnd(() => {
      if (zone.value === 1) runOnJS(fireLeft)();
      else if (zone.value === -1) runOnJS(fireRight)();
    })
    .onFinalize(() => {
      zone.value = 0;
      x.value = animateTo(0, 'snappy', reduced);
    });

  const row = useAnimatedStyle(() => ({ transform: [{ translateX: x.value }] }));
  const leftPane = useAnimatedStyle(() => ({ opacity: interpolate(x.value, [0, THRESHOLD], [0, 1], Extrapolation.CLAMP) }));
  const rightPane = useAnimatedStyle(() => ({ opacity: interpolate(x.value, [-THRESHOLD, 0], [1, 0], Extrapolation.CLAMP) }));

  return (
    <View style={{ overflow: 'hidden' }}>
      {left && (
        <Animated.View style={[{ position: 'absolute', inset: 0, backgroundColor: left.bg, justifyContent: 'center', paddingLeft: 24 }, leftPane]}>
          <Icon name={left.icon} color={left.fg} />
        </Animated.View>
      )}
      {right && (
        <Animated.View
          style={[
            { position: 'absolute', inset: 0, backgroundColor: right.bg, justifyContent: 'center', alignItems: 'flex-end', paddingRight: 24 },
            rightPane,
          ]}
        >
          <Icon name={right.icon} color={right.fg} />
        </Animated.View>
      )}
      <GestureDetector gesture={pan}>
        <Animated.View style={row}>{children}</Animated.View>
      </GestureDetector>
    </View>
  );
}
```

```tsx
import { View } from 'react-native';
import { Gesture, GestureDetector } from 'react-native-gesture-handler';
import Animated, { Extrapolation, interpolate, runOnJS, useAnimatedStyle, useReducedMotion, useSharedValue } from 'react-native-reanimated';
import { Icon, T } from '@/components/ui';
import { feedback } from '@/feedback';
import { useTheme } from '@/theme';
import { animateTo } from '@/theme/motion';

const TRACK = 280;
const KNOB = 60;
const PAD = 4;
const MAX = TRACK - KNOB - PAD * 2;

export function SlideToAnswer({ onAnswer, label = 'Slide to answer' }: { onAnswer: () => void; label?: string }) {
  const { c } = useTheme();
  const reduced = useReducedMotion();
  const x = useSharedValue(0);
  const step = useSharedValue(0);
  const done = useSharedValue(false);

  const answer = () => {
    feedback('call.answered');
    onAnswer();
  };

  const pan = Gesture.Pan()
    .onUpdate((e) => {
      if (done.value) return;
      x.value = Math.min(MAX, Math.max(0, e.translationX));
      const s = Math.floor((x.value / MAX) * 3);
      if (s !== step.value) {
        step.value = s;
        if (s > 0 && s < 3) runOnJS(feedback)('swipe.threshold');
      }
    })
    .onEnd(() => {
      if (!done.value && x.value > MAX * 0.85) {
        done.value = true;
        x.value = animateTo(MAX, 'snappy', reduced);
        runOnJS(answer)();
      } else if (!done.value) {
        x.value = animateTo(0, 'snappy', reduced);
      }
      step.value = 0;
    });

  const knob = useAnimatedStyle(() => ({ transform: [{ translateX: x.value }] }));
  const hint = useAnimatedStyle(() => ({ opacity: interpolate(x.value, [0, MAX * 0.6], [1, 0], Extrapolation.CLAMP) }));

  return (
    <View
      accessible
      accessibilityRole="button"
      accessibilityLabel={label}
      accessibilityHint="Double tap to answer"
      accessibilityActions={[{ name: 'activate' }]}
      onAccessibilityAction={(e) => e.nativeEvent.actionName === 'activate' && answer()}
      style={{ width: TRACK, height: KNOB + PAD * 2, borderRadius: (KNOB + PAD * 2) / 2, backgroundColor: c.callControl, justifyContent: 'center' }}
    >
      <Animated.View style={[{ position: 'absolute', left: KNOB + PAD * 2, right: 16, alignItems: 'center' }, hint]}>
        <T v="label" color={c.onCallScreen2}>
          {label}
        </T>
      </Animated.View>
      <GestureDetector gesture={pan}>
        <Animated.View
          style={[
            { position: 'absolute', left: PAD, width: KNOB, height: KNOB, borderRadius: KNOB / 2, backgroundColor: c.call, alignItems: 'center', justifyContent: 'center' },
            knob,
          ]}
        >
          <Icon name="call" size={28} color={c.onCall} />
        </Animated.View>
      </GestureDetector>
    </View>
  );
}
```
